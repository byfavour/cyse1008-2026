export async function POST(req) {
  const event = await req.json();
  console.log('[CALLBACK] event received =', event);

  const eventType = event.type || 'unknown';
  const orderId   = event.orderId || 'n/a';

  let responseMessage = 'Event noted';
  if (eventType === 'payment.completed') {
    responseMessage = `Order ${orderId} marked as paid — stock updated`;
  }

  return NextResponse.json({
    received: true,
    eventType,
    orderId,
    message: responseMessage,
    processedAt: new Date().toISOString(),
  });
}

Step 4 — Fetching products from Firestore

Open src/app/api/assignment5/products/route.js and read how it fetches documents and loops through them:

export async function GET() {
  const db = getDb();

  // 1. Ask Firestore for all documents in the 'products' collection
  const snapshot = await db.collection('products').get();

  // 2. Convert the snapshot into a plain array
  const products = snapshot.docs.map((doc) => ({
    id: doc.id,
    ...doc.data(),
  }));

  // 3. Loop through the array to calculate summary numbers
  let totalStock = 0;
  let totalValue = 0;

  for (const product of products) {
    const stock = Number(product.stock ?? 0);
    const price = Number(product.price ?? 0);
    totalStock += stock;
    totalValue += stock * price;
  }

  // 4. Return the summary alongside the list
  return NextResponse.json({
    count: products.length,
    totalStock,
    totalValue: Number(totalValue.toFixed(2)),
    products: products.map((p) => ({
      id:    p.id,
      name:  p.name || p.title || 'Untitled',
      price: Number(p.price ?? 0),
      stock: Number(p.stock ?? 0),
    })),
  });
}

Make sure the emulators are running, then click "Call GET /api/assignment5/products".

  • Answer: this route uses getDb() from firebase-admin. The browser pages use the regular Firebase client SDK instead. Why does an API route use the Admin SDK rather than the client SDK?
  • Answer: what does snapshot.docs.map(...) do? What kind of value does it produce?
  • Answer: the for loop adds up stock * price for every product. What ecommerce term describes this total?
